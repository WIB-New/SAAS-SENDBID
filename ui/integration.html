Exemples d'intégration

# 1. Créer un compte
curl -X POST BASE_URL/v1/accounts/register \
  -H "Content-Type: application/json" \
  -d '{"email":"you@app.com","password":"MonMotDePasse","company_name":"Mon App"}'

# 2. Upload d'un avatar
curl -X POST BASE_URL/v1/files \
  -H "X-API-Key: VOTRE_CLE_API" \
  -H "X-User-Id: user-123" \
  -F "file=@photo.jpg" \
  -F "category=avatars"

# 3. Upload KYC (chiffré automatiquement)
curl -X POST BASE_URL/v1/files \
  -H "X-API-Key: VOTRE_CLE_API" \
  -H "X-User-Id: user-123" \
  -F "file=@id_card.pdf" \
  -F "category=kyc" \
  -F 'tags={"doc_type":"id_card"}'

# 4. Lister les fichiers (paginé)
curl -H "X-API-Key: VOTRE_CLE_API" \
  BASE_URL/v1/files?page=1\&limit=20

# 5. URL signée (accès temporaire 60s)
curl -H "X-API-Key: VOTRE_CLE_API" \
  BASE_URL/v1/files/FILE_ID/signed?ttl=60

# 6. Voir le quota
curl -H "X-API-Key: VOTRE_CLE_API" \
  BASE_URL/v1/accounts/usage
# pip install httpx
import httpx

BASE = "BASE_URL"
API_KEY = "VOTRE_CLE_API"
HEADERS = {
    "X-API-Key":   API_KEY,
    "X-User-Id":   "user-123",
    "X-User-Role": "user",
}

# Upload avatar
with open("photo.jpg", "rb") as f:
    resp = httpx.post(
        f"{BASE}/v1/files",
        headers=HEADERS,
        files={"file": ("photo.jpg", f, "image/jpeg")},
        data={"category": "avatars", "tags": '{"user_id":"123"}'}
    )
result = resp.json()
print(result["url"])           # URL publique de l'avatar
print(result["thumbnail_url"])  # Miniature 256×256
print(result["file_id"])       # ID pour download/delete/signed URL

# Vérifier le quota
usage = httpx.get(f"{BASE}/v1/accounts/usage", headers=HEADERS).json()
print(f"Utilisé: {usage['storage']['used_label']} / {usage['storage']['quota_label']}")
print(f"Alerte: {usage['alerts']['quota_warning']}")

<?php
define('STORAGEHUB_URL', 'BASE_URL');
define('API_KEY', 'VOTRE_CLE_API');

// Upload d'un fichier
function uploadFile($filePath, $category, $userId) {
    $curl = curl_init();
    curl_setopt_array($curl, [
        CURLOPT_URL            => STORAGEHUB_URL . '/v1/files',
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_POST           => true,
        CURLOPT_HTTPHEADER     => [
            'X-API-Key: '   . API_KEY,
            'X-User-Id: '   . $userId,
            'X-User-Role: user',
        ],
        CURLOPT_POSTFIELDS => [
            'file'     => new CURLFile($filePath),
            'category' => $category,
        ],
    ]);
    $response = curl_exec($curl);
    curl_close($curl);
    return json_decode($response, true);
}

// Exemple d'utilisation
$result = uploadFile('/var/www/uploads/photo.jpg', 'avatars', 'user-123');
echo $result['url'] . PHP_EOL;

// Vérifier le quota
function getUsage() {
    $curl = curl_init(STORAGEHUB_URL . '/v1/accounts/usage');
    curl_setopt_array($curl, [
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_HTTPHEADER => ['X-API-Key: ' . API_KEY],
    ]);
    return json_decode(curl_exec($curl), true);
}

$usage = getUsage();
echo "Utilisé: " . $usage['storage']['used_label'];
?>
