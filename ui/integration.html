3 étapes pour intégrer votre application

Aucune infrastructure requise — votre ordinateur fait office de serveur de stockage sécurisé.

1

Créer un compte

Inscrivez-vous en bas de cette page. Vous recevez immédiatement une X-API-Key et un HMAC secret.

2

Envoyer des fichiers

Un simple POST /v1/files multipart avec votre clé en header. Le fichier est scanné, stocké et indexé automatiquement.

3

Accéder & partager

Récupérez les fichiers privés via URL signée (60 s par défaut) ou les avatars via URL CDN publique permanente.

Exemples de code

Prêt à intégrer en 5 minutes

Exemples complets dans les langages les plus courants. Copiez, adaptez, c'est prêt.

# 1. Créer un compte → obtenir vos credentials curl -X POST BASE_URL/v1/accounts/register \ -H "Content-Type: application/json" \ -d '{"email":"app@example.com","password":"MonMotDePasse","plan":"starter"}' # 2. Upload d'un avatar (public) curl -X POST BASE_URL/v1/files \ -H "X-API-Key: VOTRE_CLE_API" \ -H "X-User-Id: user-123" \ -F "file=@photo.jpg" \ -F "category=avatars" # 3. Upload KYC (chiffré AES-256 automatiquement) curl -X POST BASE_URL/v1/files \ -H "X-API-Key: VOTRE_CLE_API" \ -H "X-User-Id: user-123" \ -F "file=@id_card.pdf" \ -F "category=kyc" \ -F 'tags={"doc_type":"cni","country":"FR"}' # 4. URL signée temporaire (60 secondes) curl -H "X-API-Key: VOTRE_CLE_API" \ BASE_URL/v1/files/FILE_ID/signed?ttl=60 # 5. Quota temps réel curl -H "X-API-Key: VOTRE_CLE_API" \ BASE_URL/v1/accounts/usage
# pip install httpx import httpx BASE = "BASE_URL" API_KEY = "VOTRE_CLE_API" HEADERS = { "X-API-Key": API_KEY, "X-User-Id": "user-123", "X-User-Role": "user", } # ── Upload avatar ────────────────────────────── with open("photo.jpg", "rb") as f: r = httpx.post( f"{BASE}/v1/files", headers=HEADERS, files={"file": ("photo.jpg", f, "image/jpeg")}, data={"category": "avatars"}, ) d = r.json() print(d["url"]) # URL publique CDN print(d["thumbnail_url"]) # Miniature 256×256 file_id = d["file_id"] # ── URL signée (60 s) ────────────────────────── signed = httpx.get( f"{BASE}/v1/files/{file_id}/signed", headers=HEADERS, params={"ttl": 60} ).json()["signed_url"] # ── Quota en temps réel ──────────────────────── usage = httpx.get(f"{BASE}/v1/accounts/usage", headers=HEADERS).json() pct = usage["storage"]["used_percent"] print(f"{usage['storage']['used_label']} / {usage['storage']['quota_label']} ({pct}%)") if usage["alerts"]["quota_warning"]: print("⚠ Quota bientôt atteint !")
// SDK officiel → sdk/javascript/storagehub.js import { StorageHub } from './sdk/javascript/storagehub.js'; const hub = new StorageHub({ apiKey: "VOTRE_CLE_API", baseUrl: "BASE_URL", userId: "user-123", }); // ── Upload depuis <input type="file"> ────────── const result = await hub.upload(file, "avatars", { tags: { user_id: "123", source: "mobile" } }); console.log(result.url); // URL publique console.log(result.thumbnail_url); // Miniature auto // ── Quota ────────────────────────────────────── const { storage, alerts } = await hub.request('GET', '/v1/accounts/usage'); const pct = storage.used_percent; // ── Webhook — notifier votre backend ─────────── await hub.createWebhook("https://myapp.com/hook", "file.uploaded,file.deleted"); // ── Bulk upload ──────────────────────────────── const bulk = await hub.uploadBulk([file1, file2], "receipts"); console.log(`${bulk.uploaded} ok, ${bulk.failed} erreurs`);
<?php define('SH_URL', 'BASE_URL'); define('SH_API_KEY', 'VOTRE_CLE_API'); // ── Upload ───────────────────────────────────── function shUpload($path, $category, $userId = 'user-1') { $ch = curl_init(); curl_setopt_array($ch, [ CURLOPT_URL => SH_URL . '/v1/files', CURLOPT_RETURNTRANSFER => true, CURLOPT_POST => true, CURLOPT_HTTPHEADER => [ 'X-API-Key: ' . SH_API_KEY, 'X-User-Id: ' . $userId, 'X-User-Role: user', ], CURLOPT_POSTFIELDS => [ 'file' => new CURLFile($path), 'category' => $category, ], ]); $r = json_decode(curl_exec($ch), true); curl_close($ch); return $r; } $res = shUpload('/var/www/upload/photo.jpg', 'avatars'); echo $res['url']; // URL publique CDN echo $res['thumbnail_url']; // Miniature 256×256
